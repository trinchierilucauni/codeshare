CREATE TABLE abilita (
    id_abilita INTEGER PRIMARY KEY,
    nome VARCHAR,
    effetto VARCHAR,
    statistica VARCHAR,
    tipo VARCHAR
);

CREATE TABLE mosse (
    id_mossa INTEGER PRIMARY KEY,
    nome VARCHAR,
    potenza INTEGER,
    precisione INTEGER,
    tipo VARCHAR,
    categoria VARCHAR,
    stat_modificata VARCHAR,
    moltiplicatore NUMERIC
);

CREATE TABLE stage (
    id_stage INTEGER PRIMARY KEY,
    nome_luogo VARCHAR,
    tipo VARCHAR,
    path_img VARCHAR
);

CREATE TABLE utenti (
    nome_utente VARCHAR PRIMARY KEY,
    email VARCHAR,
    password VARCHAR(255),
    squadra INTEGER[] DEFAULT '{495,498,501,NULL,NULL,NULL}'
);

CREATE TABLE pokemon_base (
    id_pokemon INTEGER PRIMARY KEY,
    nome VARCHAR,
    ps_base INTEGER,
    attacco_base INTEGER,
    difesa_base INTEGER,
    velocita_base INTEGER,
    tipo1 VARCHAR,
    tipo2 VARCHAR,
    id_pre_evoluzione INTEGER REFERENCES pokemon_base(id_pokemon),
    id_post_evoluzione INTEGER REFERENCES pokemon_base(id_pokemon),
    sprite_fronte VARCHAR,
    sprite_retro VARCHAR,
    sprite_icon VARCHAR,
    leggendario BOOLEAN
);

CREATE TABLE strumenti (
    id_strumento INTEGER PRIMARY KEY,
    nome VARCHAR,
    effetto NUMERIC,
    statistica VARCHAR,
    tipo VARCHAR,
    id_mt INTEGER REFERENCES mosse(id_mossa),
    item_icon VARCHAR,
    descrizione VARCHAR
);

CREATE TABLE obiettivi (
    id_obiettivo INTEGER PRIMARY KEY,
    nome VARCHAR,
    descrizione VARCHAR,
    id_pokemon_sbloccato INTEGER REFERENCES pokemon_base(id_pokemon),
    tipo_azione VARCHAR,
    valore_target INTEGER
);

CREATE TABLE compatibilita_mosse (
    id_pokemon INTEGER REFERENCES pokemon_base(id_pokemon),
    id_mossa INTEGER REFERENCES mosse(id_mossa)
);

CREATE TABLE incontri_erba (
    id_stage INTEGER REFERENCES stage(id_stage),
    id_pokemon INTEGER REFERENCES pokemon_base(id_pokemon),
    percentuale_comparsa NUMERIC,
    CONSTRAINT constraint_2 PRIMARY KEY (id_stage, id_pokemon)
);

CREATE TABLE raccogli (
    id_stage INTEGER REFERENCES stage(id_stage),
    id_strumento INTEGER REFERENCES strumenti(id_strumento),
    percentuale_ritrovamento NUMERIC,
    CONSTRAINT constraint_1 PRIMARY KEY (id_stage, id_strumento)
);

CREATE TABLE allenatori_percorso (
    id_stage INTEGER REFERENCES stage(id_stage),
    id_pokemon INTEGER REFERENCES pokemon_base(id_pokemon),
    powerup_allenatore INTEGER,
    mossa1 INTEGER REFERENCES mosse(id_mossa),
    mossa2 INTEGER REFERENCES mosse(id_mossa),
    mossa3 INTEGER REFERENCES mosse(id_mossa),
    mossa4 INTEGER REFERENCES mosse(id_mossa),
    id_abilita INTEGER REFERENCES abilita(id_abilita),
    id_strumento INTEGER REFERENCES strumenti(id_strumento),
    probabilita INTEGER,
    CONSTRAINT primry_key_allenatori_percorso PRIMARY KEY (id_stage, id_pokemon)
);

CREATE TABLE boss (
    id_boss INTEGER,
    id_stage INTEGER REFERENCES stage(id_stage),
    id_pokemon INTEGER REFERENCES pokemon_base(id_pokemon),
    attacco_boss INTEGER,
    difesa_boss INTEGER,
    velocita_boss INTEGER,
    mossa1 INTEGER REFERENCES mosse(id_mossa),
    mossa2 INTEGER REFERENCES mosse(id_mossa),
    mossa3 INTEGER REFERENCES mosse(id_mossa),
    mossa4 INTEGER REFERENCES mosse(id_mossa),
    id_abilita INTEGER REFERENCES abilita(id_abilita),
    fase_due BOOLEAN,
    nome_boss VARCHAR,
    ricompensa INTEGER,
    cura BOOLEAN DEFAULT false,
    CONSTRAINT primary_key_boss PRIMARY KEY (id_boss, id_pokemon, id_stage)
);

CREATE TABLE pokedex (
    nome_utente VARCHAR REFERENCES utenti(nome_utente),
    id_pokemon INTEGER REFERENCES pokemon_base(id_pokemon),
    visto BOOLEAN,
    catturato BOOLEAN,
    CONSTRAINT utente_pokemon PRIMARY KEY (nome_utente, id_pokemon)
);

CREATE TABLE progressi_obiettivi (
    nome_utente VARCHAR REFERENCES utenti(nome_utente),
    id_obiettivo INTEGER REFERENCES obiettivi(id_obiettivo),
    valore_corrente INTEGER,
    CONSTRAINT primary_key_progressi_obiettivi PRIMARY KEY (nome_utente, id_obiettivo)
);

CREATE TABLE run_attiva (
    id_run SERIAL PRIMARY KEY,
    nome_utente VARCHAR REFERENCES utenti(nome_utente),
    soldi_attuali INTEGER,
    id_stage_corrente INTEGER REFERENCES stage(id_stage),
    tempo_trascorso INTEGER
);

CREATE TABLE pokemon_run (
    id_run INTEGER PRIMARY KEY REFERENCES run_attiva(id_run),
    id_pokemon INTEGER REFERENCES pokemon_base(id_pokemon),
    ps_rimanenti INTEGER,
    powerup_attacco INTEGER,

    powerup_velocita INTEGER,
    mossa1 INTEGER REFERENCES mosse(id_mossa),

    mossa3 INTEGER REFERENCES mosse(id_mossa),
    mossa4 INTEGER REFERENCES mosse(id_mossa),
    id_abilita_passiva INTEGER REFERENCES abilita(id_abilita),
    strumento1 INTEGER,
    strumento2 INTEGER,
    strumento3 INTEGER,
    soldi INTEGER,
    powerup_ps INTEGER DEFAULT 0
);

CREATE TABLE sala_onore (
    id_record SERIAL PRIMARY KEY,
    nome_utente VARCHAR,
    id_pokemon_finale INTEGER,
    tempo_completamento INTEGER
);
